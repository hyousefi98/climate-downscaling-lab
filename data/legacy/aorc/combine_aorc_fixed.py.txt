"""
FIXED: Combine yearly AORC zarr files into single multi-year file.

This version avoids Dask distributed timeout issues by:
1. NOT using distributed Client (uses default threaded scheduler)
2. Writing year-by-year to avoid memory issues
"""

import shutil
import numpy as np
import xarray as xr
import zarr
from pathlib import Path
from numcodecs import Blosc

# ================= CONFIGURATION =================
IN_DIR = Path("/scr/hyousefi/corrdiff_datasets/paperdata")
OUT_ZARR = Path("/scr/hyousefi/corrdiff_datasets/paperdata/aorc_3km_2000_2012.zarr")

# Change this pattern to match your files
FILE_PATTERN = "aorc_3km_{year}.zarr"

YEARS = list(range(2000, 2013))

# Chunking Config
TIME_CHUNK = 240
CHANNEL_CHUNK = 8

COMPRESSOR = Blosc(cname='lz4', clevel=5, shuffle=Blosc.SHUFFLE)

#%%
def combine_years_sequential():
    """Combine years one at a time to avoid memory/timeout issues."""
    print("=" * 60)
    print("COMBINING YEARLY ZARR FILES (SEQUENTIAL VERSION)")
    print("=" * 60)

    # Clean previous output
    if OUT_ZARR.exists():
        print(f"Removing old file: {OUT_ZARR}")
        shutil.rmtree(OUT_ZARR)

    # 1. First pass: get info about all files
    print("\n[1] Scanning yearly datasets...")
    file_info = []
    total_time = 0

    for year in YEARS:
        path = IN_DIR / FILE_PATTERN.format(year=year)
        if not path.exists():
            print(f"  ❌ {year}: File not found at {path}")
            continue

        ds = xr.open_zarr(path)
        n_steps = len(ds.time)
        total_time += n_steps

        file_info.append({
            'year': year,
            'path': path,
            'n_steps': n_steps,
            'shape': ds['aorc'].shape
        })
        print(f"  ✓ {year}: {n_steps} timesteps, shape={ds['aorc'].shape}")
        ds.close()

    if not file_info:
        print("ERROR: No datasets found!")
        return

    print(f"\n  Total timesteps: {total_time}")

    # 2. Get reference info from first file
    ref_ds = xr.open_zarr(file_info[0]['path'])
    y_size = ref_ds.sizes['y']
    x_size = ref_ds.sizes['x']
    n_channels = ref_ds.sizes['channel']

    # Get coordinates
    lat = ref_ds['lat'].values if 'lat' in ref_ds else None
    lon = ref_ds['lon'].values if 'lon' in ref_ds else None
    aorc_var = ref_ds['aorc_variable'].values if 'aorc_variable' in ref_ds else None
    ref_ds.close()

    print(f"\n[2] Creating output zarr...")
    print(f"  Shape: ({total_time}, {n_channels}, {y_size}, {x_size})")
    print(f"  Chunks: ({TIME_CHUNK}, {CHANNEL_CHUNK}, {y_size}, {x_size})")

    # 3. Create output zarr structure
    store = zarr.DirectoryStore(str(OUT_ZARR))
    root = zarr.group(store, overwrite=True)

    # Create main data array
    aorc_arr = root.create_dataset(
        'aorc',
        shape=(total_time, n_channels, y_size, x_size),
        chunks=(TIME_CHUNK, CHANNEL_CHUNK, y_size, x_size),
        dtype='float32',
        compressor=COMPRESSOR
    )
    aorc_arr.attrs['_ARRAY_DIMENSIONS'] = ['time', 'channel', 'y', 'x']

    # 4. Write year by year
    print("\n[3] Writing data year by year...")
    time_offset = 0
    all_times = []

    for info in file_info:
        year = info['year']
        path = info['path']
        n_steps = info['n_steps']

        print(f"  {year}: Writing {n_steps} timesteps (offset={time_offset})...")

        ds = xr.open_zarr(path)

        # Collect time values
        all_times.append(ds.time.values)

        # Write data in chunks to avoid memory issues
        chunk_size = TIME_CHUNK * 4  # Process multiple chunks at once
        for start in range(0, n_steps, chunk_size):
            end = min(start + chunk_size, n_steps)
            data_chunk = ds['aorc'].isel(time=slice(start, end)).values
            aorc_arr[time_offset + start : time_offset + end] = data_chunk

        ds.close()
        time_offset += n_steps

    # 5. Write coordinates
    print("\n[4] Writing coordinates...")

    # Time
    time_combined = np.concatenate(all_times)
    time_arr = root.create_dataset('time', data=time_combined.astype('datetime64[ns]').astype('int64'))
    time_arr.attrs['_ARRAY_DIMENSIONS'] = ['time']
    time_arr.attrs['units'] = 'nanoseconds since 1970-01-01'
    time_arr.attrs['calendar'] = 'proleptic_gregorian'

    # Lat/Lon
    if lat is not None:
        lat_arr = root.create_dataset('lat', data=lat)
        lat_arr.attrs['_ARRAY_DIMENSIONS'] = ['y', 'x']

    if lon is not None:
        lon_arr = root.create_dataset('lon', data=lon)
        lon_arr.attrs['_ARRAY_DIMENSIONS'] = ['y', 'x']

    # Variable names
    if aorc_var is not None:
        var_arr = root.create_dataset('aorc_variable', data=aorc_var)
        var_arr.attrs['_ARRAY_DIMENSIONS'] = ['channel']

    # Add coordinates attribute to main array
    aorc_arr.attrs['coordinates'] = 'lat lon aorc_variable'

    # Consolidate metadata
    zarr.consolidate_metadata(store)

    print(f"\n✅ Done! Output: {OUT_ZARR}")

    # 6. Verify
    print("\n[5] Verifying output...")
    ds_check = xr.open_zarr(OUT_ZARR)
    print(f"  Shape: {ds_check['aorc'].shape}")
    print(f"  Time range: {ds_check.time.values[0]} to {ds_check.time.values[-1]}")

    # Check the problematic index
    target_time = np.datetime64("2012-04-16T06:00:00")
    time_vals = ds_check.time.values.astype("datetime64[s]")
    matches = np.where(time_vals == target_time)[0]

    if len(matches) > 0:
        idx = matches[0]
        data = ds_check['aorc'].isel(time=idx, channel=5).values
        nan_count = np.isnan(data).sum()
        valid_count = np.sum(~np.isnan(data))

        print(f"\n  Checking 2012-04-16T06:00:00 (idx={idx}):")
        print(f"    Valid pixels: {valid_count}")
        print(f"    NaN pixels: {nan_count}")

        if valid_count > 0:
            print(f"    ✅ Data is valid!")
        else:
            print(f"    ❌ Still has issues!")
    else:
        print("  Target time not found in output (might be different file pattern)")

    ds_check.close()


if __name__ == "__main__":
    # NO distributed client - just run directly
    combine_years_sequential()
