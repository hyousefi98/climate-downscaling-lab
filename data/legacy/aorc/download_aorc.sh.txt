YEAR_LIST="2000 2001 2002 2003 2004 2005 2006 2007 2008 2009 2010 2011 2012"


VARS="APCP_surface DLWRF_surface DSWRF_surface latitude longitude PRES_surface SPFH_2maboveground time TMP_2maboveground UGRD_10maboveground VGRD_10maboveground .zattrs .zgroup .zmetadata"

for YEAR in $YEAR_LIST; do
  for VAR in $VARS; do
    echo "Downloading ${VAR} for year ${YEAR}..."
    aws s3 cp --no-sign-request --recursive \
      s3://noaa-nws-aorc-v1-1-1km/${YEAR}.zarr/${VAR}/ \
      /scr/hyousefi/aorc/${YEAR}.zarr/${VAR}/
  done
done
