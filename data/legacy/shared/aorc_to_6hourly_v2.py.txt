#!/usr/bin/env python3
"""
SEQUENTIAL BATCH CONVERTER (Hourly -> 6-Hourly)
-----------------------------------------------
FIXED: 
1. Loosened 'aorc_variable' check (handles DataVars vs Coords).
2. Added safety check for metadata consolidation.
3. CHUNKING: Only chunks 'time', leaves lat/lon chunks untouched.
"""

from pathlib import Path
import shutil
import numpy as np
import xarray as xr
from dask.distributed import Client
from numcodecs import Blosc
import zarr

# ================= CONFIGURATION =================
BASE_DIR = Path("/scr/hyousefi/corrdiff_datasets/paperdata")

DATASETS_TO_PROCESS = {
    # AORC Datasets
    "aorc_3km_2000_2012.zarr":          {"type": "stacked", "precip_var": "APCP_surface"},
    "aorc_4km_2000_2012.zarr":          {"type": "stacked", "precip_var": "APCP_surface"},
    "aorc_12km_2_3km_2000_2012.zarr":   {"type": "stacked", "precip_var": "APCP_surface"},
    "aorc_28km_2_4km_2000_2012.zarr":   {"type": "stacked", "precip_var": "APCP_surface"},
}

YEARS = list(range(2000, 2013))

# Output Chunking (Only Time is forced, Space is left alone)
CHUNKS_6H = {"time": 240} 
COMPRESSOR = Blosc(cname='zstd', clevel=3, shuffle=Blosc.SHUFFLE)

# ================= FUNCTIONS =================

def process_stacked_chunk(ds_chunk, precip_name):
    """
    Processes a single year of AORC data.
    """
    # --- FIX 1: Look in the WHOLE dataset (vars + coords) ---
    if "aorc_variable" in ds_chunk:
        var_list = ds_chunk["aorc_variable"].values
        var_list = [v.decode("utf-8") if isinstance(v, bytes) else v for v in var_list]
    else:
        # Debugging: Print what IS available
        print(f"    Available vars: {list(ds_chunk.variables)}")
        raise ValueError("Missing 'aorc_variable' in dataset.")

    processed_arrays = []
    da_main = ds_chunk["aorc"]
    
    for i, var_name in enumerate(var_list):
        da_chan = da_main.isel(channel=i).drop_vars(["aorc_variable", "channel"], errors="ignore")
        
        resampler = da_chan.resample(time="6h", label="right", closed="right")
        
        if var_name == precip_name:
            da_6h = resampler.sum(dim="time")
        else:
            da_6h = resampler.mean(dim="time")
            
        processed_arrays.append(da_6h)

    combined = xr.concat(processed_arrays, dim="channel")
    ds_out = xr.Dataset({"aorc": combined})
    
    ds_out = ds_out.assign_coords(
        aorc_variable=("channel", np.array(var_list, dtype="S64"))
    )
    
    for coord in ["lat", "lon"]:
        if coord in ds_chunk:
            ds_out[coord] = ds_chunk[coord]
            
    return ds_out

def process_flat_chunk(ds_chunk, precip_name):
    """
    Processes a single year of Flat data.
    """
    data_vars = list(ds_chunk.data_vars)
    resampled_vars = []
    
    for v in data_vars:
        da = ds_chunk[v]
        resampler = da.resample(time="6h", label="right", closed="right")
        
        if v == precip_name:
            da_6h = resampler.sum(dim="time")
        else:
            da_6h = resampler.mean(dim="time")
            
        resampled_vars.append(da_6h)
        
    ds_out = xr.merge(resampled_vars)
    
    if "lat" in ds_chunk.coords and "lat" not in ds_out.coords:
        ds_out = ds_out.assign_coords(lat=ds_chunk.lat, lon=ds_chunk.lon)
        
    return ds_out

def clean_encoding(ds):
    """Wipes encoding to prevent chunk conflicts."""
    for var in ds.variables:
        ds[var].encoding = {}
    return ds

# ================= MAIN =================

def main():
    client = Client()
    print(f"Cluster started: {client.dashboard_link}")
    
    for filename, config in DATASETS_TO_PROCESS.items():
        in_path = BASE_DIR / filename
        out_name = filename.replace(".zarr", "_6h.zarr")
        out_path = BASE_DIR / out_name
        
        print(f"\n{'='*60}")
        print(f"PROCESSING: {filename}")
        print(f"OUTPUT:     {out_name}")
        print(f"{'='*60}")
        
        if not in_path.exists():
            print(f"⚠️ Skipped: {in_path} does not exist.")
            continue
            
        if out_path.exists():
            print(f"⚠️ Removing existing output: {out_path}")
            shutil.rmtree(out_path)

        # Open without 'consolidated' if it causes issues, but True is usually faster
        ds_full = xr.open_zarr(in_path, consolidated=True)
        
        success_flag = False

        for i, year in enumerate(YEARS):
            print(f"  Processing {year}...", end=" ", flush=True)
            
            try:
                ds_year = ds_full.sel(time=str(year))
                
                if ds_year.sizes['time'] == 0:
                    print(f"[Skipping empty year]")
                    continue

                if config["type"] == "stacked":
                    ds_6h = process_stacked_chunk(ds_year, config["precip_var"])
                else:
                    ds_6h = process_flat_chunk(ds_year, config["precip_var"])
                
                ds_6h = clean_encoding(ds_6h)
                
                # --- CHANGED: Chunk only over time ---
                ds_6h = ds_6h.chunk(CHUNKS_6H)
                
                if i == 0:
                    # FIRST YEAR: Mode 'w'
                    print("[Writing New File]...", end=" ")
                    encoding = {v: {"compressor": COMPRESSOR, "dtype": "float32", "_FillValue": np.nan} 
                                for v in ds_6h.data_vars}
                    
                    ds_6h.to_zarr(out_path, mode='w', encoding=encoding, compute=True)
                    
                else:
                    # SUBSEQUENT YEARS: Mode 'a', SAFE_CHUNKS=FALSE
                    print("[Appending]...", end=" ")
                    
                    vars_to_drop = ["lat", "lon", "aorc_variable", "x", "y"]
                    ds_append = ds_6h.drop_vars(vars_to_drop, errors="ignore")
                    
                    # safe_chunks=False
                    ds_append.to_zarr(
                        out_path, 
                        mode='a', 
                        append_dim='time', 
                        compute=True, 
                        safe_chunks=False
                    )
                
                print("✓ Done.")
                success_flag = True
                
            except Exception as e:
                print(f"\n❌ Error processing {year}: {e}")

        # --- FIX 2: Only consolidate if file exists ---
        if success_flag and out_path.exists():
            print(f"  Consolidating metadata for {out_name}...")
            zarr.consolidate_metadata(str(out_path))
            print(f"✅ Finished {out_name}")
        else:
            print(f"❌ Failed to generate {out_name} (No years processed successfully).")

if __name__ == "__main__":
    main()
