#!/usr/bin/env python3
"""
SEQUENTIAL BATCH CONVERTER (Hourly -> 6-Hourly)
-----------------------------------------------
Uses numpy-based resampling (nanmean/nansum) to prevent NaN introduction.
Writes directly to pre-allocated zarr arrays (no xr.to_zarr append mode).
Timestamps use left-label (00:00, 06:00, 12:00, 18:00) to match WRF.
"""

from pathlib import Path
import shutil
import numpy as np
import xarray as xr
import zarr
from numcodecs import Blosc

# ================= CONFIGURATION =================
BASE_DIR = Path("/scr/hyousefi/corrdiff_datasets/paperdata")

DATASETS_TO_PROCESS = {
    "conus404_4km_2000_2012.zarr":      {"type": "flat",    "precip_var": "PREC_ACC_NC"},
    "era5_4km_2000_2012.zarr":          {"type": "flat",    "precip_var": "PREC_ACC_NC"},
}

YEARS = list(range(2000, 2013))

# Chunking Config (full spatial extent, no spatial sub-chunking)
TIME_CHUNK_6H = 240
COMPRESSOR = Blosc(cname='zstd', clevel=3, shuffle=Blosc.SHUFFLE)

# Fill value threshold
FILL_VALUE_THRESHOLD = -32000


# ================= FUNCTIONS =================

def clean_data(data):
    """Replaces fill values (<-32000) with NaN."""
    return np.where(data > FILL_VALUE_THRESHOLD, data, np.nan)


def resample_var_to_6h(data, is_precip):
    """
    Resample a single variable from hourly to 6-hourly using numpy.

    Args:
        data: numpy array (time, y, x)
        is_precip: if True, sum over 6h; otherwise mean

    Returns:
        data_6h: resampled array (n_6h, y, x)
    """
    n_time = data.shape[0]
    n_6h = n_time // 6
    n_used = n_6h * 6

    # Reshape: (n_6h*6, y, x) -> (n_6h, 6, y, x)
    data_reshaped = data[:n_used].reshape(n_6h, 6, *data.shape[1:])

    if is_precip:
        with np.errstate(invalid='ignore'):
            result = np.nansum(data_reshaped, axis=1)
            all_nan = np.all(np.isnan(data_reshaped), axis=1)
            result[all_nan] = 0.0
    else:
        with np.errstate(invalid='ignore'):
            result = np.nanmean(data_reshaped, axis=1)

    return result.astype(np.float32)


def process_flat_dataset(filename, config):
    """
    Process a flat dataset (separate variables) using direct zarr writing.
    Same approach as the working AORC combine_and_resample_6h.py.
    """
    in_path = BASE_DIR / filename
    out_name = filename.replace(".zarr", "_6h.zarr")
    out_path = BASE_DIR / out_name
    precip_var = config["precip_var"]

    print(f"\n{'='*60}")
    print(f"PROCESSING: {filename}")
    print(f"OUTPUT:     {out_name}")
    print(f"{'='*60}")

    if not in_path.exists():
        print(f"  Skipped: {in_path} does not exist.")
        return

    if out_path.exists():
        print(f"  Removing existing output: {out_path}")
        shutil.rmtree(out_path)

    # 1. Scan all years to get total size
    print("\n[1] Scanning years...")
    ds_full = xr.open_zarr(in_path, consolidated=True)

    total_hours = 0
    year_info = []
    for year in YEARS:
        ds_year = ds_full.sel(time=str(year))
        n_hours = ds_year.sizes['time']
        if n_hours == 0:
            continue
        total_hours += n_hours
        year_info.append({'year': year, 'n_hours': n_hours})
        print(f"  {year}: {n_hours} hours")

    total_6h = total_hours // 6
    print(f"\n  Total hours: {total_hours}")
    print(f"  Total 6-hourly: {total_6h}")

    # 2. Get metadata
    print("\n[2] Reading metadata...")
    # Get variable names (only time-dependent data vars)
    var_names = [v for v in ds_full.data_vars if 'time' in ds_full[v].dims]
    n_y = ds_full.sizes['y']
    n_x = ds_full.sizes['x']

    # Get lat/lon
    lat = ds_full['lat'].values if 'lat' in ds_full.coords else None
    lon = ds_full['lon'].values if 'lon' in ds_full.coords else None

    print(f"  Spatial shape: ({n_y}, {n_x})")
    print(f"  Variables ({len(var_names)}): {var_names}")
    print(f"  Precip var: {precip_var}")

    # 3. Create output zarr with pre-allocated arrays
    print("\n[3] Creating output zarr...")
    store = zarr.DirectoryStore(str(out_path))
    root = zarr.group(store, overwrite=True)

    # Create a zarr array for each variable
    zarr_arrays = {}
    for var_name in var_names:
        arr = root.create_dataset(
            var_name,
            shape=(total_6h, n_y, n_x),
            chunks=(TIME_CHUNK_6H, n_y, n_x),
            dtype='float32',
            compressor=COMPRESSOR,
            fill_value=np.nan
        )
        arr.attrs['_ARRAY_DIMENSIONS'] = ['time', 'y', 'x']
        zarr_arrays[var_name] = arr
        print(f"  Created {var_name}: ({total_6h}, {n_y}, {n_x})")

    # 4. Process each year
    print("\n[4] Processing years...")
    out_idx = 0
    all_times_6h = []

    for info in year_info:
        year = info['year']
        n_hours = info['n_hours']
        n_6h_year = n_hours // 6

        print(f"\n  {year}: {n_hours} hours -> {n_6h_year} 6h steps")

        ds_year = ds_full.sel(time=str(year))
        times = ds_year.time.values
        n_used = n_6h_year * 6

        # Left-label timestamps (00:00, 06:00, 12:00, 18:00) matching WRF
        times_6h = times[:n_used:6]
        all_times_6h.append(times_6h)

        # Process each variable
        for var_name in var_names:
            print(f"    {var_name}...", end=" ", flush=True)

            # Load one variable at a time
            data = ds_year[var_name].values[:n_used]
            data = clean_data(data)

            # Resample
            data_6h = resample_var_to_6h(
                data, is_precip=(var_name == precip_var)
            )

            # Write directly to zarr
            zarr_arrays[var_name][out_idx:out_idx + n_6h_year] = data_6h

            del data, data_6h
            print("done", flush=True)

        out_idx += n_6h_year
        print(f"  {year} done (wrote to idx {out_idx - n_6h_year}:{out_idx})")

    # 5. Write coordinates
    print("\n[5] Writing coordinates...")

    times_6h_combined = np.concatenate(all_times_6h)
    print(f"  Total 6h timesteps: {len(times_6h_combined)}")

    # Time
    time_arr = root.create_dataset(
        'time',
        data=times_6h_combined.astype('datetime64[ns]').astype('int64')
    )
    time_arr.attrs['_ARRAY_DIMENSIONS'] = ['time']
    time_arr.attrs['units'] = 'nanoseconds since 1970-01-01'
    time_arr.attrs['calendar'] = 'proleptic_gregorian'

    # Lat/Lon
    if lat is not None:
        lat_arr = root.create_dataset('lat', data=lat)
        lat_arr.attrs['_ARRAY_DIMENSIONS'] = list(ds_full['lat'].dims)

    if lon is not None:
        lon_arr = root.create_dataset('lon', data=lon)
        lon_arr.attrs['_ARRAY_DIMENSIONS'] = list(ds_full['lon'].dims)

    # Consolidate metadata
    zarr.consolidate_metadata(store)

    ds_full.close()

    # 6. Verify
    print("\n[6] Verifying output...")
    ds_check = xr.open_zarr(out_path)
    print(f"  Time range: {ds_check.time.values[0]} to {ds_check.time.values[-1]}")
    print(f"  Total 6h steps: {len(ds_check.time)}")
    print(f"  Variables: {list(ds_check.data_vars)}")

    # Spot check for NaN
    for var_name in var_names[:3]:
        for tidx in [0, len(ds_check.time)//2, len(ds_check.time)-1]:
            data = ds_check[var_name].isel(time=tidx).values
            nan_pct = 100 * np.isnan(data).sum() / data.size
            print(f"  {var_name} idx={tidx}: NaN%={nan_pct:.1f}%")

    ds_check.close()
    print(f"\n  Finished {out_name}")


def process_stacked_dataset(filename, config):
    """
    Process a stacked (AORC-style) dataset using direct zarr writing.
    Same approach as the working AORC combine_and_resample_6h.py.
    """
    in_path = BASE_DIR / filename
    out_name = filename.replace(".zarr", "_6h.zarr")
    out_path = BASE_DIR / out_name
    precip_var = config["precip_var"]

    print(f"\n{'='*60}")
    print(f"PROCESSING: {filename}")
    print(f"OUTPUT:     {out_name}")
    print(f"{'='*60}")

    if not in_path.exists():
        print(f"  Skipped: {in_path} does not exist.")
        return

    if out_path.exists():
        print(f"  Removing existing output: {out_path}")
        shutil.rmtree(out_path)

    # 1. Scan all years to get total size
    print("\n[1] Scanning years...")
    ds_full = xr.open_zarr(in_path, consolidated=True)

    total_hours = 0
    year_info = []
    for year in YEARS:
        ds_year = ds_full.sel(time=str(year))
        n_hours = ds_year.sizes['time']
        if n_hours == 0:
            continue
        total_hours += n_hours
        year_info.append({'year': year, 'n_hours': n_hours})
        print(f"  {year}: {n_hours} hours")

    total_6h = total_hours // 6
    print(f"\n  Total hours: {total_hours}")
    print(f"  Total 6-hourly: {total_6h}")

    # 2. Get metadata
    print("\n[2] Reading metadata...")
    n_channel = ds_full.sizes['channel']
    n_y = ds_full.sizes['y']
    n_x = ds_full.sizes['x']

    if "aorc_variable" in ds_full:
        var_names = [v.decode("utf-8") if isinstance(v, bytes) else str(v)
                     for v in ds_full["aorc_variable"].values]
    else:
        raise ValueError("Dataset missing 'aorc_variable'")

    lat = ds_full['lat'].values if 'lat' in ds_full else None
    lon = ds_full['lon'].values if 'lon' in ds_full else None

    print(f"  Shape per timestep: ({n_channel}, {n_y}, {n_x})")
    print(f"  Variables: {var_names}")

    # 3. Create output zarr
    print("\n[3] Creating output zarr...")
    store = zarr.DirectoryStore(str(out_path))
    root = zarr.group(store, overwrite=True)

    aorc_arr = root.create_dataset(
        'aorc',
        shape=(total_6h, n_channel, n_y, n_x),
        chunks=(TIME_CHUNK_6H, n_channel, n_y, n_x),
        dtype='float32',
        compressor=COMPRESSOR,
        fill_value=np.nan
    )
    aorc_arr.attrs['_ARRAY_DIMENSIONS'] = ['time', 'channel', 'y', 'x']
    aorc_arr.attrs['coordinates'] = 'lat lon aorc_variable'

    # 4. Process each year
    print("\n[4] Processing years...")
    out_idx = 0
    all_times_6h = []

    for info in year_info:
        year = info['year']
        n_hours = info['n_hours']
        n_6h_year = n_hours // 6

        print(f"\n  {year}: {n_hours} hours -> {n_6h_year} 6h steps")

        ds_year = ds_full.sel(time=str(year))
        times = ds_year.time.values
        n_used = n_6h_year * 6

        times_6h = times[:n_used:6]
        all_times_6h.append(times_6h)

        data_6h = np.zeros((n_6h_year, n_channel, n_y, n_x), dtype=np.float32)

        for i, var_name in enumerate(var_names):
            print(f"    {var_name}...", end=" ", flush=True)

            channel_data = ds_year['aorc'].isel(channel=i).values[:n_used]
            channel_data = clean_data(channel_data)

            data_6h[:, i, :, :] = resample_var_to_6h(
                channel_data, is_precip=(var_name == precip_var)
            )
            del channel_data
            print("done", flush=True)

        aorc_arr[out_idx:out_idx + n_6h_year] = data_6h
        out_idx += n_6h_year
        del data_6h
        print(f"  {year} done (wrote to idx {out_idx - n_6h_year}:{out_idx})")

    # 5. Write coordinates
    print("\n[5] Writing coordinates...")

    times_6h_combined = np.concatenate(all_times_6h)
    print(f"  Total 6h timesteps: {len(times_6h_combined)}")

    time_arr = root.create_dataset(
        'time',
        data=times_6h_combined.astype('datetime64[ns]').astype('int64')
    )
    time_arr.attrs['_ARRAY_DIMENSIONS'] = ['time']
    time_arr.attrs['units'] = 'nanoseconds since 1970-01-01'
    time_arr.attrs['calendar'] = 'proleptic_gregorian'

    if lat is not None:
        lat_arr = root.create_dataset('lat', data=lat)
        lat_arr.attrs['_ARRAY_DIMENSIONS'] = ['y', 'x']

    if lon is not None:
        lon_arr = root.create_dataset('lon', data=lon)
        lon_arr.attrs['_ARRAY_DIMENSIONS'] = ['y', 'x']

    var_arr = root.create_dataset('aorc_variable', data=np.array(var_names, dtype='S64'))
    var_arr.attrs['_ARRAY_DIMENSIONS'] = ['channel']

    zarr.consolidate_metadata(store)
    ds_full.close()

    print(f"\n  Finished {out_name}")


# ================= MAIN =================

def main():
    for filename, config in DATASETS_TO_PROCESS.items():
        if config["type"] == "stacked":
            process_stacked_dataset(filename, config)
        else:
            process_flat_dataset(filename, config)

    print("\n" + "=" * 60)
    print("ALL DONE!")
    print("=" * 60)


if __name__ == "__main__":
    main()
