def get_lat_lon_from_dataset(ds):
    """Extract lat/lon arrays from reference grid dataset."""

    # Check for 2D lat/lon
    if 'LAT2D' in ds and 'LON2D' in ds:
        lat_2d = ds['LAT2D'].values
        lon_2d = ds['LON2D'].values
        print(f"  Using LAT2D/LON2D (shape: {lat_2d.shape})")
    elif 'lat' in ds.coords and 'lon' in ds.coords:
        lat = ds.coords['lat'].values
        lon = ds.coords['lon'].values

        # If 1D, create 2D meshgrid
        if lat.ndim == 1 and lon.ndim == 1:
            lon_2d, lat_2d = np.meshgrid(lon, lat)
            print(f"  Created meshgrid from 1D lat/lon (shape: {lat_2d.shape})")
        else:
            lat_2d = lat
            lon_2d = lon
            print(f"  Using 2D lat/lon (shape: {lat_2d.shape})")
    else:
        raise ValueError("Could not find lat/lon coordinates in dataset")

    return lat_2d, lon_2d

# STEP 3: Create Land/Ocean Mask Using Cartopy
# ==============================================================================

def create_land_mask_cartopy(lat_2d, lon_2d, resolution='10m'):
    """
    Create land/ocean mask using Cartopy Natural Earth features.

    Parameters:
    -----------
    lat_2d : 2D array
        Latitude coordinates
    lon_2d : 2D array
        Longitude coordinates
    resolution : str
        Natural Earth resolution ('10m', '50m', or '110m')

    Returns:
    --------
    mask : 2D array
        Boolean mask where True = Land, False = Ocean
    """
    print(f"\nCreating land mask using Natural Earth {resolution} resolution...")
    print(f"  Grid shape: {lat_2d.shape}")
    print(f"  Total points: {lat_2d.size:,}")

    # Load Natural Earth land geometries
    land_shp = shapereader.natural_earth(resolution=resolution,
                                         category='physical',
                                         name='land')
    land_reader = shapereader.Reader(land_shp)
    land_geoms = list(land_reader.geometries())

    print(f"  Loaded {len(land_geoms)} land polygons")

    # Create mask array
    mask = np.zeros(lat_2d.shape, dtype=bool)

    # Check each point
    total_points = lat_2d.size
    land_count = 0

    print("  Processing grid points...")
    for i in range(lat_2d.shape[0]):
        if i % 10 == 0:  # Progress update every 10 rows
            progress = (i * lat_2d.shape[1]) / total_points * 100
            print(f"    Progress: {progress:.1f}% ({land_count:,} land points so far)")

        for j in range(lat_2d.shape[1]):
            point = Point(lon_2d[i, j], lat_2d[i, j])

            # Check if point is in any land polygon
            for geom in land_geoms:
                if geom.contains(point):
                    mask[i, j] = True
                    land_count += 1
                    break

    ocean_count = total_points - land_count
    print(f"\n  Mask Statistics (before expansion):")
    print(f"    Total points: {total_points:,}")
    print(f"    Land points:  {land_count:,} ({land_count/total_points*100:.2f}%)")
    print(f"    Ocean points: {ocean_count:,} ({ocean_count/total_points*100:.2f}%)")

    # Expand ocean by dilating the ocean mask (or equivalently, eroding the land mask)
    # This adds 1 cell buffer around ocean boundaries
    print("\n  Expanding ocean mask by 1 pixel (coastal buffer)...")
    from scipy.ndimage import binary_dilation

    # Create ocean mask (inverse of land)
    ocean_mask = ~mask

    # Dilate ocean: expands ocean areas by 1 pixel into land
    expanded_ocean_mask = binary_dilation(ocean_mask, iterations=1)

    # Convert back to land mask
    final_land_mask = ~expanded_ocean_mask

    # Calculate new statistics
    final_land_count = final_land_mask.sum()
    final_ocean_count = total_points - final_land_count
    coastal_cells_added_to_ocean = land_count - final_land_count

    print(f"  Mask Statistics (after expansion):")
    print(f"    Land points:  {final_land_count:,} ({final_land_count/total_points*100:.2f}%)")
    print(f"    Ocean points: {final_ocean_count:,} ({final_ocean_count/total_points*100:.2f}%)")
    print(f"    Coastal cells added to ocean: {coastal_cells_added_to_ocean:,}")

    return final_land_mask
